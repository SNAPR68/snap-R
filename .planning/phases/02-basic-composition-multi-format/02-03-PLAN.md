**ADD render status polling useEffect (recursive setTimeout pattern from research):**
```typescript
useEffect(() => {
  if (!renderId || renderStatus === 'completed' || renderStatus === 'failed') return

  let isMounted = true

  const poll = async () => {
    try {
      const response = await fetch(`/api/video/status?renderId=${renderId}`, {
        signal: AbortSignal.timeout(15000),
      })
      const data = await response.json()

      if (!isMounted) return

      if (data.status === 'completed' && data.videoUrl) {
        setVideoUrl(data.videoUrl)
        setRenderStatus('completed')
        setProgress(100)
        setProgressMessage('Video ready!')
        setGenerating(false)
        trackEvent(SnapREvents.VIDEO_CREATED)
      } else if (data.status === 'failed') {
        setRenderError(data.error || 'Render failed')
        setRenderStatus('failed')
        setGenerating(false)
        setProgressMessage('')
      } else {
        // Still rendering — update progress and continue polling
        const progressPercent = Math.round((data.progress ?? 0) * 100)
        setProgress(progressPercent)
        setProgressMessage(`Rendering video... ${progressPercent}%`)
        setTimeout(poll, 3000) // Poll every 3 seconds
      }
    } catch {
      // Network error — retry with longer interval
      if (isMounted) {
        setTimeout(poll, 5000)
      }
    }
  }

  poll()

  return () => {
    isMounted = false
  }
}, [renderId, renderStatus])
```
